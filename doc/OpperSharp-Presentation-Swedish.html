Problemet: REST API Komplexitet


var httpClient = new HttpClient {
    BaseAddress = new Uri("https://api.opper.ai")
};
httpClient.DefaultRequestHeaders.Authorization =
    new AuthenticationHeaderValue("Bearer", apiKey);

var requestBody = new { input = new { question = "Vad är AI?" } };
var content = new StringContent(
    JsonConvert.SerializeObject(requestBody),
    Encoding.UTF8, "application/json"
);

var response = await httpClient.PostAsync("/v1/call/func", content);
var responseString = await response.Content.ReadAsStringAsync();

if (!response.IsSuccessStatusCode) {
    throw new Exception($"API-fel: {responseString}");
}

var result = JsonConvert.DeserializeObject<dynamic>(responseString);

~50 rader boilerplate-kod!

Lösningen: OpperSharp SDK


var client = OpperClient.FromEnvironment();

var response = await client.CallAsync(
    path: "func",
    input: new Dictionary<string, object> {
        ["question"] = "Vad är AI?"
    }
);

3 rader. 90% mindre kod!

  • ✅ Automatisk felhantering
  • ✅ Typsäkerhet & IntelliSense
  • ✅ Inbyggd återförsökslogik

2. Indexes API

Bygg Kunskapsbaser


// Skapa index
var index = await client.Indexes.CreateAsync(
    name: "företagsdokument",
    description: "Intern dokumentation"
);

// Lägg till dokument (bulk)
await client.Indexes.AddBulkAsync("företagsdokument", new[] {
    new OpperDocument {
        Content = "Semesterpolicy: 20 dagar per år",
        Metadata = new() { ["category"] = "HR" }
    },
    new OpperDocument {
        Content = "Kodgranskning kräver 2 godkännanden",
        Metadata = new() { ["category"] = "Engineering" }

5. Agent-Ramverk


public class MathTools
{
    [Tool("Adderar två tal")]
    public double Add(double a, double b) => a + b;

    [Tool("Multiplicerar två tal")]
    public double Multiply(double a, double b) => a * b;

    [Tool("Beräknar procent")]
    public double Percentage(double value, double percent)
        => value * (percent / 100);
}

Attribut-baserad verktygsupptäckt!

Agent-Användning


var agent = new Agent(client, new AgentOptions {
    FunctionPath = "math-solver",
    MaxIterations = 10
}).WithTools(new MathTools());

var response = await agent.RunAsync(
    "Om jag investerar 1000 kr med 15% avkastning, " +
    "sedan lägger till 500 kr, vad blir totalen?"
);

Console.WriteLine($"Svar: {response.Output}");
Console.WriteLine($"Iterationer: {response.Iterations}");
                    
Svar: Din totala summa blir 1650 kr
Iterationer: 3
Verktyg använda: 3

Verkligt Exempel: RAG-Pipeline


// Steg 1: Sök kunskapsbas
var searchResults = await client.Indexes.QueryAsync(
    indexName: "företagsdokument",
    query: "Vad är vår kodgranskningsprocess?",
    k: 3
);

// Steg 2: Bygg kontext
var context = string.Join("\n\n",
    searchResults.Results.Select(r => r.Content));

// Steg 3: Generera svar med kontext
var answer = await client.Chat.CompleteAsync(
    prompt: "Vad är vår kodgranskningsprocess?",
    systemPrompt: $@"Svara endast med denna kontext:
{context}
Om inte i kontext, säg 'Jag har inte den informationen.'"
);

Console.WriteLine(answer);

Användningsfall: Kundsupport-Bot


async Task<string> HandleCustomerQuery(string query)
{
    // Sök kunskapsbas
    var docs = await client.Indexes.QueryAsync(
        "hjälpdokument", query, k: 5
    );

    var context = string.Join("\n",
        docs.Results.Select(r => r.Content)
    );

    // Generera svar
    return await client.Chat.CompleteAsync(
        query,
        systemPrompt: $"Använd denna kontext:\n{context}"
    );
}
