Kör dessa kommandon i C# för att debugga Opper API:

// Test 1: Embeddings (kräver inte functions)
var embedding = await client.Embeddings.CreateAsync(
    "Hello world",
    "azure/text-embedding-3-small"
);
Console.WriteLine($"Embeddings fungerar! Dimensioner: {embedding.Count}");

// Test 2: Lista alla functions
var functions = await client.Functions.ListAsync();
Console.WriteLine($"Antal functions: {functions.Count}");
foreach (var func in functions)
{
    Console.WriteLine($"  - {func.Path}");
}

// Test 3: Testa en specifik function
try
{
    var response = await client.Functions.CallAsync(
        "math-solver",
        new Dictionary<string, object> { ["problem"] = "What is 2 + 2?" },
        new OpperCallOptions { Model = "gpt-4" }
    );
    Console.WriteLine($"Response: {response.Message}");
}
catch (Exception ex)
{
    Console.WriteLine($"Error: {ex.Message}");
}
